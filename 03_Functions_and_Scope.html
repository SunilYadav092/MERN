JavaScript Functions & Scope

Functions are reusable blocks of code, and Scope determines variable visibility.

1. Function Declarations

Different Ways to Declare Functions

// Function declaration
function greet(name) {
    return "Hello, " + name;
}

// Function expression
const add = function(a, b) {
    return a + b;
};

// Arrow function (ES6)
const multiply = (a, b) => a * b;

// Arrow function with multiple statements
const calculate = (a, b) => {
    let sum = a + b;
    return sum * 2;
};

// Immediately Invoked Function Expression (IIFE)
(function() {
    console.log("This runs immediately!");
})();

2. Scope

Global, Function, and Block Scope

// Global scope
let globalVar = "I'm global";

function myFunction() {
    // Function scope
    let funcVar = "I'm in function";
    
    if (true) {
        // Block scope (let and const)
        let blockVar = "I'm in block";
        var oldVar = "I'm function scoped";
    }
    
    console.log(funcVar);       // Works
    console.log(blockVar);      // ReferenceError!
}

console.log(globalVar);         // Works
console.log(myFunction);        // Function not accessible

3. Closures

Understanding Closures

// Closure example
function createCounter() {
    let count = 0;      // This variable persists
    
    return function() {
        count++;
        return count;
    };
}

const counter = createCounter();
console.log(counter());  // 1
console.log(counter());  // 2
console.log(counter());  // 3

// The inner function "closes over" the count variable

4. Higher-Order Functions

Functions that Return or Accept Functions

// Function that returns a function
function makeMultiplier(factor) {
    return function(number) {
        return number * factor;
    };
}

const double = makeMultiplier(2);
const triple = makeMultiplier(3);

console.log(double(5));  // 10
console.log(triple(5));  // 15

// Function that accepts a function
function processArray(arr, callback) {
    for (let i = 0; i < arr.length; i++) {
        arr[i] = callback(arr[i]);
    }
    return arr;
}

let nums = [1, 2, 3];
nums = processArray(nums, n => n * 2);

6. Default Parameters & Rest Parameters

Modern Function Features

// Default parameters
function greet(name = "Guest", greeting = "Hello") {
    return greeting + ", " + name;
}

console.log(greet());           // Hello, Guest
console.log(greet("John"));     // Hello, John

// Rest parameters
function sum(...numbers) {
    let total = 0;
    for (let num of numbers) {
        total += num;
    }
    return total;
}

console.log(sum(1, 2, 3));      // 6
console.log(sum(1, 2, 3, 4, 5)); // 15

// Combining default and rest
function formatList(prefix = "Items:", ...items) {
    return prefix + " " + items.join(", ");
}
