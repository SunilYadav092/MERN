Modern JavaScript ES6+ Features

ES6+ (ECMAScript 2015+) introduced many modern features that make JavaScript more powerful and expressive.

1. Template Literals

String Interpolation with Backticks

// Old way
let name = "John";
let age = 30;
let msg = "My name is " + name + " and I'm " + age;

// ES6 Template Literals
const message = `My name is ${name} and I'm ${age}`;

// Multi-line strings
const bio = `
    Name: ${name}
    Age: ${age}
    Hobby: Coding
`;

// Expression evaluation
const total = `Total: ${5 + 3}`;  // Total: 8

3. Arrow Functions

Concise Function Syntax

// Regular function
function add(a, b) {
    return a + b;
}

// Arrow function (concise)
const add = (a, b) => a + b;

// Arrow function with single parameter
const square = x => x * x;

// Arrow function with no parameters
const greet = () => "Hello!";

// Arrow function with multiple statements
const calculate = (a, b) => {
    const sum = a + b;
    const product = a * b;
    return {sum, product};
};

// Arrow functions and this binding
const person = {
    name: "John",
    greet: function() {
        return `Hello, I'm ${this.name}`;  // Works
    },
    arrow: () => {
        return this.name;  // Doesn't work - arrow functions don't have 'this'
    }
};

4. Default Parameters

Function Default Values

// Old way
function greet(name) {
    name = name || "Guest";
    return "Hello, " + name;
}

// ES6 Default Parameters
const greetEs6 = (name = "Guest", greeting = "Hello") => {
    return `${greeting}, ${name}`;
};

// With expressions
const getConfig = (timeout = 5000, retries = 3) => {
    return {timeout, retries};
};

// Can reference other parameters
const createURL = (domain, path = "/", protocol = "https") => {
    return `${protocol}://${domain}${path}`;
};

5. Rest & Spread Operators

... Operator for Arrays & Objects

// Spread with arrays
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const combined = [...arr1, ...arr2];  // [1,2,3,4,5,6]

// Spread with objects
const obj1 = {a: 1, b: 2};
const obj2 = {c: 3, d: 4};
const merged = {...obj1, ...obj2};

// Rest parameters
function sum(...numbers) {
    return numbers.reduce((a, b) => a + b, 0);
}
sum(1, 2, 3, 4, 5);  // 15

// Rest in destructuring
const [first, second, ...rest] = [1, 2, 3, 4, 5];
// first=1, second=2, rest=[3,4,5]

6. Enhanced Object Literals

Shorthand & Computed Properties

// Old way
const name = "John";
const age = 30;
const person = {
    name: name,
    age: age,
    greet: function() {
        return "Hello";
    }
};

// ES6 Shorthand
const name = "John";
const age = 30;
const person = {
    name,           // Same as name: name
    age,            // Same as age: age
    greet() {       // Same as greet: function()
        return "Hello";
    }
};

// Computed properties
const key = "dynamicKey";
const obj = {
    [key]: "value",
    ["computed_" + Math.random()]: 42

7. For...of Loop

Iterating Over Values

const arr = ["a", "b", "c"];
const obj = {x: 1, y: 2, z: 3};

// For...of - iterates over values
for (const value of arr) {
    console.log(value);  // "a", "b", "c"
}

// For...in - iterates over keys
for (const key in obj) {
    console.log(key, obj[key]);  // x 1, y 2, z 3
}

// With destructuring
for (const [index, value] of arr.entries()) {
    console.log(index, value);
}

// String iteration
for (const char of "hello") {
    console.log(char);
}

8. Modules (Import/Export)

Module System

// math.js
export const add = (a, b) => a + b;
export const multiply = (a, b) => a * b;
export default class Calculator { /* ... */ }

// main.js - Importing
import Calculator, {add, multiply} from './math.js';

// Alternative import syntax
import * as math from './math.js';
console.log(math.add(5, 3));

// CommonJS (Node.js)
module.exports = {add, multiply};
const {add, multiply} = require('./math');

9. Promise & Async/Await

Asynchronous Code

// Promises (ES6)
const promise = new Promise((resolve, reject) => {
    if (success) resolve(data);
    else reject(error);
});

promise.then(data => {/* ... */})
       .catch(error => {/* ... */});

// Async/Await (ES2017)
async function fetchData() {
    try {
        const response = await fetch('/api/data');
        const data = await response.json();
        return data;
    } catch (error) {
        console.log("Error:", error);
    }
}

// Multiple await
async function loadAll() {
    const users = await getUsers();
    const posts = await getPosts();
    return {users, posts};

10. Optional Chaining & Nullish Coalescing

Safe Property Access (ES2020+)

const user = {name: "John", address: {city: "NYC"}};

// Optional chaining ?.
user?.name;              // "John"
user?.age;               // undefined (safe)
user.address?.zip;       // undefined (safe)
user.profile?.avatar?.url;  // Safe even if profile doesn't exist

// Optional chaining with function calls
user.getName?.();        // Calls if method exists
arr?.[0];                // Safe array access

// Nullish coalescing ??
const value = null ?? "default";  // "default"
const count = 0 ?? "no count";    // 0 (0 is not null/undefined)

// Compared to ||
const value2 = null || "default";  // "default"
const count2 = 0 || "no count";    // "no count" (0 is falsy)
