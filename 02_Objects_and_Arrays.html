JavaScript Objects & Arrays

Objects and Arrays are fundamental data structures in JavaScript for storing and organizing data.

1. Arrays

Array Creation & Access

// Creating arrays
let fruits = ["Apple", "Banana", "Orange"];
let numbers = [1, 2, 3, 4, 5];
let mixed = [1, "Hello", true, {name: "John"}];

// Accessing elements
console.log(fruits[0]);        // Apple
console.log(fruits.length);    // 3

// Methods
fruits.push("Mango");          // Add to end
fruits.pop();                  // Remove from end
fruits.unshift("Cherry");      // Add to start
fruits.shift();                // Remove from start

Array Methods

// Iteration methods
let nums = [1, 2, 3, 4, 5];

nums.forEach(n => console.log(n));
nums.map(n => n * 2);          // Transform each element
nums.filter(n => n > 2);       // Keep elements matching condition
nums.reduce((sum, n) => sum + n, 0);  // Accumulate value

// Search methods
nums.includes(3);              // true
nums.indexOf(3);               // 2
nums.find(n => n > 3);         // 4
nums.some(n => n > 3);         // true

2. Objects

Object Creation & Properties

// Object literal
let person = {
    firstName: "John",
    lastName: "Doe",
    age: 30,
    email: "john@example.com",
    isActive: true,
    skills: ["JavaScript", "Python", "HTML"]
};

// Accessing properties
person.firstName;              // John
person["firstName"];           // John (bracket notation)

// Adding properties
person.phone = "555-1234";

// Deleting properties
delete person.isActive;

// Checking properties
"firstName" in person;         // true
person.hasOwnProperty("age");  // true

Object Methods

let person = {name: "Alice", age: 25, city: "Boston"};

// Get keys, values, entries
Object.keys(person);           // ["name", "age", "city"]
Object.values(person);         // ["Alice", 25, "Boston"]
Object.entries(person);        // [["name", "Alice"], ...]

// Clone/merge objects
let clone = {...person};       // Spread operator
let merged = Object.assign({}, person, {age: 26});

// Check properties
Object.getOwnPropertyNames(person);

4. Destructuring

Array & Object Destructuring

// Array destructuring
let [a, b, c] = [1, 2, 3];
console.log(a, b, c);          // 1, 2, 3

// Skip elements
let [x, , z] = [10, 20, 30];
console.log(x, z);             // 10, 30

// Object destructuring
let {name, age} = {name: "John", age: 30, city: "NYC"};
console.log(name, age);        // John, 30

// Renaming
let {name: personName, age: personAge} = {name: "Alice", age: 25};
