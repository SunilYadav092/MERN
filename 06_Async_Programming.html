JavaScript Asynchronous Programming

Asynchronous programming allows operations to run without blocking the main thread. Essential for API calls, timers, and file operations.

1. setTimeout & setInterval

Delayed Execution

// Execute code after delay (milliseconds)
setTimeout(() => {
    console.log("This runs after 2 seconds");
}, 2000);

// Execute code repeatedly
setInterval(() => {
    console.log("This runs every 1 second");
}, 1000);

// Clear timeout/interval
let timerId = setTimeout(...);
clearTimeout(timerId);

3. Async/Await

Modern Asynchronous Code

// Async function (always returns a promise)
async function fetchData() {
    try {
        const response = await somePromise();
        console.log(response);
        return response;
    } catch (error) {
        console.log("Error:", error);
    } finally {
        console.log("Done");
    }
}

// Calling async function
fetchData().then(result => {
    console.log("Result:", result);
});

8. Callback Hell vs Modern Async

Code Comparison

// ❌ Callback Hell (Pyramid of Doom)
function oldWay() {
    getUser(1, function(user) {
        getOrders(user.id, function(orders) {
            getOrderDetails(orders[0].id, function(details) {
                console.log(details);
            });
        });
    });
}

// ✓ Promises
function betterWay() {
    getUser(1)
        .then(user => getOrders(user.id))
        .then(orders => getOrderDetails(orders[0].id))
        .then(details => console.log(details));
}

// ✓✓ Async/Await (Best)
async function bestWay() {
    const user = await getUser(1);
    const orders = await getOrders(user.id);
    const details = await getOrderDetails(orders[0].id);
    console.log(details);
}
